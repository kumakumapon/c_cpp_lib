add_library(embed_crc src/embed_crc.c)
target_include_directories(embed_crc PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)

    add_subdirectory(tests)
endif()

    target_compile_definitions(embed_crc PRIVATE EMBED_CRC_TABLES)
endif()
